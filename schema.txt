-- ============================================================

CREATE TABLE app.organization_settings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    organization_name TEXT NOT NULL,
    legal_name TEXT,
    logo_media_id UUID,
    primary_email CITEXT,
    primary_phone TEXT,
    whatsapp_number TEXT,
    default_language VARCHAR(10) NOT NULL DEFAULT 'en',
    timezone TEXT NOT NULL DEFAULT 'Asia/Kolkata',
    primary_location_id UUID,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.organization_locations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    address_line_1 TEXT,
    address_line_2 TEXT,
    city TEXT,
    state TEXT,
    postal_code TEXT,
    country TEXT NOT NULL DEFAULT 'India',
    latitude NUMERIC(9, 6),
    longitude NUMERIC(9, 6),
    google_maps_url TEXT,
    phone TEXT,
    email CITEXT,
    office_hours TEXT,
    is_primary BOOLEAN NOT NULL DEFAULT false,
    status app.record_status NOT NULL DEFAULT 'active',
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE app.organization_settings
    ADD CONSTRAINT fk_org_primary_location
    FOREIGN KEY (primary_location_id)
    REFERENCES app.organization_locations(id)
    ON DELETE SET NULL;

-- ============================================================
-- AUTHENTICATION / RBAC
-- ============================================================

CREATE TABLE app.users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email CITEXT NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    is_super_admin BOOLEAN NOT NULL DEFAULT false,
    is_active BOOLEAN NOT NULL DEFAULT true,
    last_login_at TIMESTAMPTZ,
    password_changed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.employees (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL UNIQUE REFERENCES app.users(id) ON DELETE CASCADE,
    employee_code TEXT NOT NULL UNIQUE,
    full_name TEXT NOT NULL,
    designation TEXT,
    phone TEXT,
    status app.employee_status NOT NULL DEFAULT 'active',
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.roles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role_name TEXT NOT NULL UNIQUE,
    description TEXT,
    is_system_role BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.permissions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    permission_key TEXT NOT NULL UNIQUE,
    module_name TEXT NOT NULL,
    action_name TEXT NOT NULL,
    description TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.employee_roles (
    employee_id UUID NOT NULL REFERENCES app.employees(id) ON DELETE CASCADE,
    role_id UUID NOT NULL REFERENCES app.roles(id) ON DELETE CASCADE,
    assigned_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    assigned_by UUID REFERENCES app.users(id) ON DELETE SET NULL,
    PRIMARY KEY (employee_id, role_id)
);

CREATE TABLE app.role_permissions (
    role_id UUID NOT NULL REFERENCES app.roles(id) ON DELETE CASCADE,
    permission_id UUID NOT NULL REFERENCES app.permissions(id) ON DELETE CASCADE,
    effect app.permission_effect NOT NULL DEFAULT 'allow',
    PRIMARY KEY (role_id, permission_id)
);

CREATE TABLE app.employee_permission_overrides (
    employee_id UUID NOT NULL REFERENCES app.employees(id) ON DELETE CASCADE,
    permission_id UUID NOT NULL REFERENCES app.permissions(id) ON DELETE CASCADE,
    effect app.permission_effect NOT NULL,
    assigned_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    assigned_by UUID REFERENCES app.users(id) ON DELETE SET NULL,
    PRIMARY KEY (employee_id, permission_id)
);

CREATE TABLE app.member_categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL UNIQUE,
    description TEXT,
    display_order INTEGER NOT NULL DEFAULT 0,
    status app.record_status NOT NULL DEFAULT 'active',
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    member_code TEXT NOT NULL UNIQUE,
    category_id UUID REFERENCES app.member_categories(id) ON DELETE SET NULL,

    first_name TEXT NOT NULL,
    middle_name TEXT,
    last_name TEXT,
    display_name TEXT NOT NULL,

    designation TEXT,

    phone TEXT,
    email CITEXT,
    address_line_1 TEXT,
    address_line_2 TEXT,
    city TEXT,
    state TEXT,
    postal_code TEXT,

    current_management BOOLEAN NOT NULL DEFAULT false,
    management_post TEXT,
    management_display_order INTEGER NOT NULL DEFAULT 0,

    status app.record_status NOT NULL DEFAULT 'active',

    joined_date DATE,

    search_document TSVECTOR GENERATED ALWAYS AS (
        setweight(to_tsvector('simple', coalesce(display_name, '')), 'A') ||
        setweight(to_tsvector('simple', coalesce(designation, '')), 'B') ||
        setweight(to_tsvector('simple', coalesce(member_code, '')), 'B')
    ) STORED,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.member_visibility (
    member_id UUID PRIMARY KEY REFERENCES app.members(id) ON DELETE CASCADE,

    photo_visibility app.visibility NOT NULL DEFAULT 'public',
    designation_visibility app.visibility NOT NULL DEFAULT 'public',
    phone_visibility app.visibility NOT NULL DEFAULT 'private',
    email_visibility app.visibility NOT NULL DEFAULT 'private',
    address_visibility app.visibility NOT NULL DEFAULT 'private',
    management_visibility app.visibility NOT NULL DEFAULT 'public',

    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.member_social_links (

CREATE TABLE app.member_photos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    member_id UUID NOT NULL REFERENCES app.members(id) ON DELETE CASCADE,
    media_asset_id UUID,
    is_primary BOOLEAN NOT NULL DEFAULT false,
    display_order INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()

-- ============================================================
-- EVENTS
-- ============================================================

CREATE TABLE app.events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_code TEXT NOT NULL UNIQUE,
    title TEXT NOT NULL,
    description TEXT,

    event_date DATE NOT NULL,
    start_time TIME,
    end_time TIME,

    location_id UUID REFERENCES app.organization_locations(id) ON DELETE SET NULL,
    location_name TEXT,
    address TEXT,
    google_maps_url TEXT,

    status app.event_status NOT NULL DEFAULT 'upcoming',

    is_featured BOOLEAN NOT NULL DEFAULT false,
    is_countdown_enabled BOOLEAN NOT NULL DEFAULT false,

    record_status app.record_status NOT NULL DEFAULT 'active',

    search_document TSVECTOR GENERATED ALWAYS AS (
        setweight(to_tsvector('simple', coalesce(title, '')), 'A') ||
        setweight(to_tsvector('simple', coalesce(description, '')), 'B') ||
        setweight(to_tsvector('simple', coalesce(event_code, '')), 'B')
    ) STORED,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE app.social_work_events (
    social_work_activity_id UUID NOT NULL
        REFERENCES app.social_work_activities(id) ON DELETE CASCADE,

    event_id UUID NOT NULL
        REFERENCES app.events(id) ON DELETE CASCADE,

    PRIMARY KEY (social_work_activity_id, event_id)
);

CREATE TABLE app.event_albums (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID NOT NULL UNIQUE REFERENCES app.events(id) ON DELETE CASCADE,
    title TEXT NOT NULL,
    description TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ============================================================
-- MEDIA
-- ============================================================

CREATE TABLE media.media_assets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    original_filename TEXT NOT NULL,
    mime_type TEXT NOT NULL,
    file_size_bytes BIGINT NOT NULL,

    width INTEGER,
    height INTEGER,

    source_provider TEXT NOT NULL DEFAULT 'google_drive',
    source_file_id TEXT,
    source_path TEXT,

    status app.media_status NOT NULL DEFAULT 'pending',

    checksum_sha256 TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT positive_media_size CHECK (file_size_bytes > 0),
    CONSTRAINT positive_media_dimensions CHECK (
        (width IS NULL AND height IS NULL)
        OR
        (width > 0 AND height > 0)
    )
);

CREATE TABLE media.photo_variants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    media_asset_id UUID NOT NULL REFERENCES media.media_assets(id) ON DELETE CASCADE,

    variant_name TEXT NOT NULL,
    storage_provider TEXT NOT NULL DEFAULT 'cloudflare_r2',
    storage_key TEXT NOT NULL UNIQUE,

    width INTEGER NOT NULL,
    height INTEGER NOT NULL,
    file_size_bytes BIGINT NOT NULL,
    mime_type TEXT NOT NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    UNIQUE (media_asset_id, variant_name)
);

CREATE TABLE app.photos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    album_id UUID NOT NULL REFERENCES app.event_albums(id) ON DELETE CASCADE,

    uploaded_by UUID REFERENCES app.users(id) ON DELETE SET NULL,

    status app.record_status NOT NULL DEFAULT 'active',

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE imports.import_batches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    batch_code TEXT NOT NULL UNIQUE,

    module_name TEXT NOT NULL,
    original_filename TEXT NOT NULL,

    uploaded_by UUID REFERENCES app.users(id) ON DELETE SET NULL,

    status app.import_status NOT NULL DEFAULT 'uploaded',

    total_rows INTEGER NOT NULL DEFAULT 0,
    accepted_rows INTEGER NOT NULL DEFAULT 0,
    rejected_rows INTEGER NOT NULL DEFAULT 0,
    warning_rows INTEGER NOT NULL DEFAULT 0,

    template_version TEXT,

    started_at TIMESTAMPTZ,
    validated_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE imports.import_rows (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    batch_id UUID NOT NULL
        REFERENCES imports.import_batches(id) ON DELETE CASCADE,

    row_number INTEGER NOT NULL,

    row_status app.import_row_status NOT NULL DEFAULT 'pending',

    business_code TEXT,

    original_data JSONB NOT NULL,

    normalized_data JSONB,

    validation_messages JSONB,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    UNIQUE (batch_id, row_number)
);

CREATE TABLE imports.rejected_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    import_row_id UUID NOT NULL UNIQUE
        REFERENCES imports.import_rows(id) ON DELETE CASCADE,

    batch_id UUID NOT NULL
        REFERENCES imports.import_batches(id) ON DELETE CASCADE,

    module_name TEXT NOT NULL,

    original_data JSONB NOT NULL,

    status app.rejection_status NOT NULL DEFAULT 'rejected',

    corrected_at TIMESTAMPTZ,
    re_uploaded_at TIMESTAMPTZ,
    resolved_at TIMESTAMPTZ,

    resolved_by UUID REFERENCES app.users(id) ON DELETE SET NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE imports.rejection_errors (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    rejected_record_id UUID NOT NULL
        REFERENCES imports.rejected_records(id) ON DELETE CASCADE,

    WHERE current_management = true
      AND status = 'active';

-- ============================================================
-- IMPORTANT BUSINESS CONSTRAINTS
-- ============================================================

ALTER TABLE app.members
    ADD CONSTRAINT chk_management_post_required
    CHECK (
        current_management = false
        OR management_post IS NOT NULL
    );

ALTER TABLE app.events
    ADD CONSTRAINT chk_event_end_after_start
    CHECK (
        start_time IS NULL
        OR end_time IS NULL
        OR end_time >= start_time
    );

ALTER TABLE cms.announcements
    ADD CONSTRAINT chk_announcement_dates
    CHECK (
        expires_at IS NULL
        OR publish_at IS NULL
        OR expires_at > publish_at
    );

ALTER TABLE app.organization_locations
    ADD CONSTRAINT chk_single_primary_location
    CHECK (
        is_primary = false
        OR status = 'active'
    );

CREATE UNIQUE INDEX ux_primary_location
    ON app.organization_locations(is_primary)
    WHERE is_primary = true
      AND status = 'active';

CREATE UNIQUE INDEX ux_primary_member_photo
    ON app.member_photos(member_id)
    WHERE is_primary = true;
